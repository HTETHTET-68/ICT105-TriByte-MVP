Total Records = DISTINCTCOUNT('PrototypeRecords'[RecordID])
Total Activity Events = COUNTROWS('ActivityLog')
New Submissions = CALCULATE(COUNTROWS('ActivityLog'), 'ActivityLog'[ActionType] = "SubmitReport")
Pending Cases = CALCULATE(DISTINCTCOUNT('PrototypeRecords'[RecordID]), 'PrototypeRecords'[CurrentStatus] = "Pending")
Resolved Cases = CALCULATE(DISTINCTCOUNT('PrototypeRecords'[RecordID]), 'PrototypeRecords'[CurrentStatus] IN {"Claimed", "Returned", "Closed"})
Resolution Rate = DIVIDE([Resolved Cases], [Total Records])
Average Resolution Days = AVERAGE('PrototypeRecords'[ResolutionDays])
Search Actions = CALCULATE(COUNTROWS('ActivityLog'), 'ActivityLog'[ActionType] = "SearchRecord")
Admin Updates = CALCULATE(COUNTROWS('ActivityLog'), 'ActivityLog'[ActionType] IN {"UpdateStatus", "AdminReview"})
Dashboard Views = CALCULATE(COUNTROWS('ActivityLog'), 'ActivityLog'[ActionType] = "ViewDashboard")
Average Processing Time = AVERAGE('ActivityLog'[ProcessingTimeMinutes])
Mobile Activity Events = CALCULATE(COUNTROWS('ActivityLog'), 'ActivityLog'[DeviceType] = "Mobile")
Mobile Activity Rate = DIVIDE([Mobile Activity Events], [Total Activity Events])
